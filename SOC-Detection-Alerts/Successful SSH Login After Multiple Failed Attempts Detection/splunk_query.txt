index=main ("Failed password" OR "Accepted password")
| rex "password for (?<ssh_user>\w+)"
| rex "from (?<src_ip>\d{1,3}(?:\.\d{1,3}){3})"
| stats
count(eval(searchmatch("Failed password"))) AS failed_count
count(eval(searchmatch("Accepted password"))) AS success_count
values(src_ip) AS src_ip
values(host) AS host
by ssh_user
| where failed_count >= 3 AND success_count >= 1
